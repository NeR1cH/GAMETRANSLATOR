# 1. Внесите изменения в код
# 2. Загрузите на GitHub:
git add .
git commit -m "Описание изменений"
git push

# 3. Создайте новый тег:
git tag v1.0.1
git push origin v1.0.1

# GitHub автоматически соберёт новый .exe!
# Пересоздайте тег:
git tag -d v1.0.0# Удалите старый тег с удалённого репозитория:
git push origin --delete v1.0.0# Создайте тег заново:
git tag v1.0.0# Загрузите тег на удалённый репозиторий:
git push origin v1.0.0# Готово!
