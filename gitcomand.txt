# 1. Внесите изменения в код
# 2. Загрузите на GitHub:
git add .
git commit -m "Описание изменений"
git push

# 3. Создайте новый тег:
git tag v1.0.1
git push origin v1.0.1

# GitHub автоматически соберёт новый .exe!
# Пересоздайте тег:
# Удалите старый тег с удалённого репозитория:
git tag -d v1.0.0
# Создайте тег заново:
git push origin --delete v1.0.0
# Загрузите тег на удалённый репозиторий:
git tag v1.0.0
# Готово!
git push origin v1.0.0
